# Smoothed level and 500 ms activity hold: sustained low E near the gate, 300 ms strum gaps,
# bounded silence tail, and immediate cut on clipping, stale samples and tuner entry.
calibrate
press foot 100
expect running=true gate=.004
# Sustained low E2 at -41 dBFS (cycle RMS ~0.00436, just above the 0.004 gate) credits continuously;
# YIN does not run outside tuner / stage 3 (`analyses=0`).
tone 5000 82.4069 -41
expect active=true active_ms=5000 analyses=0
# A 2 s silence credits at most the 500 ms hold (490 ms after the final waveform trough), then stops.
tone 2000 0 -20
expect active=false hold_ms=0 active_ms=5490
tone 1000 0 -20
expect active=false active_ms=5490
# Strums with 300 ms silent chord-change gaps stay active across every gap.
tone 6000 110 -20 300 300
expect active=true hold_ms=200 active_ms=11490
# Drain the remaining 200 ms hold, then verify a fresh 1 s note + 2 s silence adds exactly 500 ms.
audio 1000 0 0
expect active=false hold_ms=0 active_ms=11690
audio 1000 .1 .3
expect active=true hold_ms=500 active_ms=12690
audio 2000 0 0
expect active=false hold_ms=0 active_ms=13190
# Clipping cuts an active hold immediately with zero hold tail and zero smoother carryover.
audio 1000 .1 .3
expect active=true hold_ms=500 active_ms=14190
audio 1000 .1 1
expect active=false hold_ms=0 active_ms=14190
audio 1000 0 0
expect active=false active_ms=14190
# Entering the tuner zeroes the hold; once in tuner mode no time is credited.
audio 1000 .1 .3
expect active=true hold_ms=500 active_ms=15190
press mode 100
expect tuning=true running=false active=false hold_ms=0 active_ms~=15190
tone 300 440 -20
expect pitch_valid=true pitch_midi=69 analyses=7 active_ms~=15190
audio 1000 0 0 440 .99
expect active_ms~=15190
serial p
press mode 100
press foot 100
# Stale samples (unplugged cable) zero the hold as soon as freshness expires (<100 ms).
audio 1000 .1 .3
expect active=true hold_ms=500
wait 1000
expect active=false hold_ms=0
